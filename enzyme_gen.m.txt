function[m1_gent,lifem1,e1_gent,lifee1,changem_t1,m1,te1,ne1]=enzyme_gen(minh,einh,ton1,tof1,ts)

te1=[];
ne1=[];
changem_t1=[];
m1=[];

totm=2000;

if sum(minh)<0
    inhm=0;
else 
   inhm=sum(minh);
end

if sum(einh)<0
    inhe=0;
else 
    inhe=sum(einh);
end

[m1_gent1]=burst(totm-inhm,ton1,tof1);
m1_gent=(ts*ones(1,length(m1_gent1)))+m1_gent1;
if inhm>0
m1_gent=[ts*ones(1,inhm),m1_gent];
end

m1g=ones(1,length(m1_gent));
m1d=-1*ones(1,length(m1_gent));
mn1=[m1g,m1d];
lifem1=exprnd(60*25,1,length(m1_gent)); %add lives here in the front
m1_dect=-(m1_gent+lifem1);
m1t=[m1_gent,m1_dect];
[mt1,idx1]=sort(abs(m1t));
mn1=mn1(idx1);
summ1=0;

for i=1:length(mn1)
    summ1=summ1+mn1(i);
    m1(i)=summ1;
    changem_t1(i)=mt1(i);
end

changem_t1=[ts,changem_t1((inhm+1):end)];
m1=[inhm,m1((inhm+1):end)];

count1=0;
for b=1:length(m1g)
    np1(b)=floor(lifem1(b)/34.3);
    if np1(b) >= 1 
        eppp1(b)=changem_t1(b);
    else 
        eppp1(b)=0;
    end
end

up1=eppp1(eppp1>0);

co=1;
for b=1:length(up1)
    gent=up1(b);
    for c=1:np1(b)
        e1_gent(co)=gent+(34.3*c);
        co=co+1;
    end
end

if sum(einh)>0
e1_gent=[ts*ones(1,sum(einh)),e1_gent];
end

e1g=ones(1,length(e1_gent));
e1d=-1*ones(1,length(e1_gent));

en1=[e1g,e1d];
lifee1=exprnd(3600*20,1,length(e1_gent));

e1_dect=-(e1_gent+lifee1);
e1t=[e1_gent,e1_dect];
[et1,idy1]=sort(abs(e1t));
en1=en1(idy1);

sume1=0;
for i=1:length(en1)
    sume1=sume1+en1(i);
    e1(i)=sume1;
    changee1_t1(i)=et1(i);
end

changee1_t1=[ts,changee1_t1((inhe+1):end)];

e1=[inhe,e1((inhe+1):end)];

[te1,Ide1]=unique(changee1_t1);
ne1=e1(Ide1);
end
